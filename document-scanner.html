let selectedFile = null; let imageURL = ""; function selectFile() { document.getElementById("fileInput").click(); } document.getElementById("fileInput").addEventListener("change", function () { selectedFile = this.files[0]; if (!selectedFile) { return; } imageURL = URL.createObjectURL(selectedFile); document.getElementById("message").innerText = "Document selected: " + selectedFile.name; document.getElementById("previewBtn").style.display = "inline-block"; document.getElementById("downloadBtn").style.display = "inline-block"; document.getElementById("previewImage").src = imageURL; document.getElementById("previewImage").style.display = "block"; }); function showPreview() { if (!selectedFile) { alert("Please select a document first."); return; } document.getElementById("previewImage").style.display = "block"; } function downloadScan() { if (!selectedFile) { alert("Please select a document first."); return; } const link = document.createElement("a"); link.href = imageURL; link.download = "EasyLife-Scanned-Document.jpg"; document.body.appendChild(link); link.click(); document.body.removeChild(link); } function cropDocument() { if (!selectedFile) { alert("Please select a document first."); return; } const image = document.getElementById("previewImage"); image.style.display = "block"; const cropArea = document.getElementById("cropArea"); cropArea.innerHTML = ""; cropArea.style.display = "block"; const cropImage = document.createElement("img"); cropImage.src = image.src; cropImage.style.width = "100%"; cropImage.style.height = "100%"; cropImage.style.objectFit = "contain"; cropArea.appendChild(cropImage); let startX = 0; let startY = 0; let selecting = false; const selection = document.createElement("div"); selection.style.position = "absolute"; selection.style.border = "2px solid red"; selection.style.background = "rgba(255,0,0,0.15)"; selection.style.display = "none"; cropArea.appendChild(selection); cropArea.onmousedown = function(e) { const rect = cropArea.getBoundingClientRect(); startX = e.clientX - rect.left; startY = e.clientY - rect.top; selecting = true; selection.style.display = "block"; selection.style.left = startX + "px"; selection.style.top = startY + "px"; selection.style.width = "0px"; selection.style.height = "0px"; }; cropArea.onmousemove = function(e) { if (!selecting) return; const rect = cropArea.getBoundingClientRect(); const currentX = e.clientX - rect.left; const currentY = e.clientY - rect.top; const width = Math.abs(currentX - startX); const height = Math.abs(currentY - startY); selection.style.left = Math.min(startX, currentX) + "px"; selection.style.top = Math.min(startY, currentY) + "px"; selection.style.width = width + "px"; selection.style.height = height + "px"; }; cropArea.onmouseup = function() { selecting = false; if ( parseInt(selection.style.width) < 20 || parseInt(selection.style.height) < 20 ) { return; } alert("Crop area selected successfully!"); }; }
